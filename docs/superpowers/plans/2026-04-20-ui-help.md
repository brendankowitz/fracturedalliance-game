```typescript
import type { DifficultyLevel } from "@fa/sim";
import { DIFFICULTY_PRESETS } from "@fa/sim";
import { useUiStore } from "../store/uiStore.ts";
import type { RenderLoopHandle } from "../game/renderLoop.ts";

interface DifficultySelectorProps {
  onDifficultySelect: (difficulty: DifficultyLevel) => void;
  renderLoopHandle: RenderLoopHandle | null;
}

const DIFFICULTY_BUTTONS: DifficultyLevel[] = ["easy", "normal", "hard", "brutal", "nightmare"];

const DIFFICULTY_DESCRIPTIONS: Record<DifficultyLevel, string> = {
  easy: "Perfect for learning. Abundant resources, lenient timers, peaceful rivals.",
  normal: "Balanced challenge. Standard economy, moderate pressure from rivals.",
  hard: "Advanced players. Scarce resources, aggressive rivals, tight margins.",
  brutal: "Veteran difficulty. Extreme scarcity, relentless attacks, high stakes.",
  nightmare: "Insane. Everything wants you dead. Resources vanish. No mercy.",
};

export function DifficultySelector({ onDifficultySelect, renderLoopHandle }: DifficultySelectorProps) {
  const selectedDifficulty = useUiStore((s) => s.selectedDifficulty);
  const setDifficulty = useUiStore((s) => s.setDifficulty);

  const handleSelect = (difficulty: DifficultyLevel) => {
    setDifficulty(difficulty);
    onDifficultySelect(difficulty);
  };

  return (
    <div
      style={{
        position: "absolute",
        inset: 0,
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        justifyContent: "center",
        background: "rgba(0,8,20,0.98)",
        color: "var(--text)",
        fontFamily: "var(--font-ui)",
        gap: 24,
        zIndex: 50,
        padding: "40px 20px",
      }}
    >
      <h2 style={{ fontSize: 24, margin: 0, letterSpacing: 2, fontFamily: "var(--font-head)" }}>
        SELECT DIFFICULTY
      </h2>

      <div style={{ display: "flex", flexDirection: "column", gap: 12, maxWidth: 600 }}>
        {DIFFICULTY_BUTTONS.map((level) => {
          const preset = DIFFICULTY_PRESETS[level];
          const isSelected = selectedDifficulty === level;
          return (
            <button
              key={level}
              type="button"
              onClick={() => handleSelect(level)}
              style={{
                background: isSelected ? "rgba(68, 136, 204, 0.2)" : "rgba(68, 136, 204, 0.05)",
                border: `2px solid ${isSelected ? "var(--blue)" : "var(--border)"}`,
                color: isSelected ? "var(--blue)" : "var(--text)",
                fontFamily: "var(--font-ui)",
                fontSize: 16,
                padding: "16px 20px",
                cursor: "pointer",
                textAlign: "left",
                transition: "all 200ms ease",
              }}
            >
              <div style={{ fontWeight: 700, fontSize: 16, marginBottom: 4 }}>
                {preset?.label}
              </div>
              <div style={{ fontSize: 13, opacity: 0.8, lineHeight: 1.4 }}>
                {DIFFICULTY_DESCRIPTIONS[level]}
              </div>
            </button>
          );
        })}
      </div>

      <button
        type="button"
        onClick={() => handleSelect(selectedDifficulty)}
        style={{
          background: "var(--blue)",
          border: "1px solid var(--blue)",
          color: "var(--bg)",
          fontFamily: "var(--font-head)",
          fontSize: 14,
          padding: "12px 40px",
          cursor: "pointer",
          letterSpacing: 1,
          fontWeight: 700,
          marginTop: 16,

  );
}
```

```typescript
import { setup } from "xstate";
import type { HudSnapshot } from "@fa/sim";

export type TutorialEvent = { type: "ADVANCE" } | { type: "DISMISS" };

interface StepConfig {
  message: string;
  position: {
    bottom?: number;
    top?: number;
    left?: number | string;
    right?: number | string;
  };
  predicate: (snap: HudSnapshot) => boolean;
}

export const TUTORIAL_STEPS: Record<string, StepConfig> = {
  step1: {
    message: "...",
    position: { bottom: 120, left: 16 },
    predicate: (snap) => {
      // Player owns an asteroid with Air Processor
      return snap.asteroids.some(
        (a) =>
          a.ownerId === snap.humanPlayerId &&
          (a.buildingKinds.includes("airProcessor") ||
            a.buildQueue.some((q) => q.buildingKind === "airProcessor")),
      );
    },
  },
  step2: {
    message: "...",
    position: { bottom: 120, left: 16 },
    predicate: (snap) => {
      // Player owns a Mine building
      return snap.asteroids.some(
        (a) =>
          a.ownerId === snap.humanPlayerId &&
          (a.buildingKinds.includes("mineMk1") ||
            a.buildQueue.some((q) => q.buildingKind === "mineMk1")),
      );
    },
  },
  step3: {
    message: "...",
    position: { top: 80, left: "calc(50% - 160px)" },
    predicate: (snap) => {
      // Player has credits > 15000
      return snap.players.some(
        (p) => p.id === snap.humanPlayerId && p.credits > 15000,
      );
    },
  },
  step4: {
    message: "...",
    position: { bottom: 120, left: 16 },
    predicate: (snap) => {
      // Player owns a blueprint
      return snap.players.some(
        (p) => p.id === snap.humanPlayerId && p.blueprints.length > 0,
      );
    },
  },
  step5: {
    message: "...",
    position: { top: 80, right: 16 },
    predicate: (snap) => {
      // Player has population > 50
      return snap.asteroids.some(
        (a) => a.ownerId === snap.humanPlayerId && (a.population ?? 0) > 50,
      );
    },
  },
};

export const tutorialMachine = setup({
  types: { events: {} as TutorialEvent },
}).createMachine({
  id: "tutorial",
  initial: "step1",
  states: {
    step1: { on: { ADVANCE: "step2", DISMISS: "dismissed" } },
    step2: { on: { ADVANCE: "step3", DISMISS: "dismissed" } },
    step3: { on: { ADVANCE: "step4", DISMISS: "dismissed" } },
    step4: { on: { ADVANCE: "step5", DISMISS: "dismissed" } },
    step5: { on: { ADVANCE: "done", DISMISS: "dismissed" } },
    done: { type: "final" },
    dismissed: { type: "final" },
  },
});
```

- [ ] **Step 2.2:** Update TutorialTooltip to check predicates before enabling ADVANCE

Modify `apps/web/src/hud/TutorialTooltip.tsx`:
```typescript
import { useMachine } from "@xstate/react";
import { useEffect, useState } from "react";
import { TUTORIAL_STEPS, tutorialMachine } from "../machines/tutorialMachine.ts";
import { useGameStore } from "../store/gameStore.ts";

export function TutorialTooltip() {
  const [state, send] = useMachine(tutorialMachine);
  const snapshot = useGameStore((s) => s.snapshot);
  const [predicateMet, setPredicateMet] = useState(false);

  useEffect(() => {
    if (!snapshot) return;
    const current = state.value as string;
    if (current === "done" || current === "dismissed") return;

    const stepConfig = TUTORIAL_STEPS[current];
    if (!stepConfig) return;

    const met = stepConfig.predicate(snapshot);
    setPredicateMet(met);

    // Auto-advance if predicate just became true
    if (met && !predicateMet) {
      send({ type: "ADVANCE" });
    }
  }, [snapshot, state.value, send, predicateMet]);

  if (state.matches("done") || state.matches("dismissed")) return null;

  const current = state.value as string;
  const stepConfig = TUTORIAL_STEPS[current];
  if (!stepConfig) return null;

  const stepNumber = Object.keys(TUTORIAL_STEPS).indexOf(current) + 1;
  const isStep5 = current === "step5";

  return (
    <div
      style={{
        position: "absolute",
        zIndex: 25,
        width: 300,
        background: "var(--bg-panel)",
        border: "1px solid var(--amber)",
        boxShadow: "0 0 20px rgba(255,146,0,0.2), inset 0 0 30px rgba(255,146,0,0.03)",
        color: "var(--text)",
        fontFamily: "var(--font-data)",
        ...stepConfig.position,
      }}
    >
      <div style={{ display: "flex", alignItems: "center", gap: 8, padding: "8px 12px", borderBottom: "1px solid rgba(255,146,0,0.2)", background: "rgba(255,146,0,0.06)" }}>
        <div className="fa-tutorial-dot" />
        <span style={{ fontFamily: "var(--font-head)", fontSize: 9, letterSpacing: 2, color: "var(--amber)", fontWeight: 700, flex: 1 }}>
          MISSION BRIEFING
        </span>
        <div style={{ display: "flex", gap: 4 }}>
          {Array.from({ length: 5 }, (_, i) => (
            <div key={i} style={{ width: 5, height: 5, borderRadius: "50%", background: i < stepNumber ? "var(--amber)" : "var(--border)" }} />
          ))}
        </div>
      </div>

      <div style={{ padding: "6px 12px 0" }}>
        <span style={{ fontFamily: "var(--font-ui)", fontSize: 10, color: "var(--amber)", letterSpacing: 1, fontWeight: 600 }}>
          STEP {stepNumber} OF 5
        </span>
      </div>

      <div style={{ padding: "6px 12px 10px", fontSize: 13, lineHeight: 1.6, color: "var(--text-hi)", fontFamily: "var(--font-ui)", fontWeight: 500 }}>
        {stepConfig.message}
      </div>

      {!predicateMet && (
        <div style={{ padding: "6px 12px", fontSize: 11, color: "var(--text-lo)", fontStyle: "italic" }}>
          Waiting for objective...
        </div>
      )}

      <div style={{ display: "flex", gap: 8, padding: "8px 12px", borderTop: "1px solid rgba(255,146,0,0.15)" }}>
        {isStep5 && predicateMet && (
          <button
            type="button"
            onClick={() => send({ type: "ADVANCE" })}
            className="fa-btn fa-btn-primary"
            style={{ flex: 1 }}
          >
            Got it
          </button>
        )}
        <button
          type="button"
          onClick={() => send({ type: "DISMISS" })}
          className="fa-btn"
          style={{ marginLeft: isStep5 && predicateMet ? 0 : "auto" }}
        >
          Skip Tutorial
        </button>
      </div>
    </div>
  );
}
```

```typescript
import { useState } from "react";

  return (
    <div style={{ position: "relative", display: "inline-block" }}>

        type="button"

          display: "inline-flex",
          alignItems: "center",
          justifyContent: "center",
          width: 16,

          border: "1px solid var(--border)",
          fontSize: 10,
          fontWeight: 700,
          cursor: "pointer",
          marginLeft: 6,
        }}

        ?
      </button>

      {showTooltip && (
        <div
          style={{
            position: "absolute",
            bottom: "calc(100% + 8px)",
            left: "50%",
            transform: "translateX(-50%)",
            background: "var(--bg-panel)",
            border: "1px solid var(--border)",
            borderRadius: 4,

}
```

```typescript
import { create } from "zustand";
import { persist } from "zustand/middleware";

export interface KeybindMap {
  pause: string;
  zoomIn: string;
  zoomOut: string;
  openEspionage: string;
  openDiplomacy: string;
  openTrade: string;
  openBlueprints: string;
  openBlackMarket: string;
}

const DEFAULT_KEYBINDS: KeybindMap = {
  pause: " ", // Space
  zoomIn: "+",
  zoomOut: "-",
  openEspionage: "e",
  openDiplomacy: "d",
  openTrade: "t",
  openBlueprints: "r",
  openBlackMarket: "m",
};

interface KeybindState {
  keybinds: KeybindMap;
  setKeybind: (action: keyof KeybindMap, key: string) => void;
  resetKeybinds: () => void;
  getKeybind: (action: keyof KeybindMap) => string;
  getDuplicates: () => Map<string, string[]>; // key -> list of actions
}

export const useKeybindStore = create<KeybindState>(
  persist(
    (set, get) => ({
      keybinds: DEFAULT_KEYBINDS,
      setKeybind: (action, key) =>
        set((s) => ({
          keybinds: { ...s.keybinds, [action]: key },
        })),
      resetKeybinds: () => set({ keybinds: DEFAULT_KEYBINDS }),
      getKeybind: (action) => get().keybinds[action],
      getDuplicates: () => {
        const dupes = new Map<string, string[]>();
        const keybinds = get().keybinds;
        const actions = Object.entries(keybinds);
        actions.forEach(([action, key]) => {
          const matching = actions
            .filter(([a]) => keybinds[a as keyof KeybindMap] === key)
            .map(([a]) => a);
          if (matching.length > 1) {
            dupes.set(key, matching);
          }
        });
        return dupes;
      },
    }),
    { name: "keybind-store", version: 1 },
  ),
);
```

```typescript
  const handleStartListen = (action: string) => {
    setListening(action);
  };

  const handleKeyDown = (e: React.KeyboardEvent, action: string) => {
    e.preventDefault();
    const key = e.key === " " ? " " : e.key.toLowerCase();
    setKeybind(action as any, key);
    setListening(null);
  };

  return (
    <div
      style={{
        display: "flex",
        flexDirection: "column",
        gap: 12,
        padding: 12,
        background: "var(--bg-panel)",
        border: "1px solid var(--border)",
      }}
    >
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
        <h3 style={{ margin: 0, fontSize: 14 }}>Keybindings</h3>
        <button
          type="button"
          onClick={resetKeybinds}
          className="fa-btn fa-btn-sm"
        >
          Reset to Defaults
        </button>
      </div>

      {Object.entries(keybinds).map(([action, key]) => {
        const isDuplicate = dupes.has(key);
        const isListening = listening === action;

        return (
          <div
            key={action}
            style={{
              display: "flex",
              justifyContent: "space-between",
              alignItems: "center",
              padding: 8,
              background: isDuplicate ? "rgba(255, 0, 0, 0.1)" : "rgba(0, 0, 0, 0.2)",
              border: `1px solid ${isDuplicate ? "var(--red)" : "var(--border)"}`,
              borderRadius: 4,
            }}
          >
            <label style={{ fontSize: 12, fontWeight: 500 }}>
              {ACTION_LABELS[action]}
            </label>
            <input
              type="text"
              readOnly={!isListening}
              value={isListening ? "Press a key..." : key}
              onKeyDown={(e) => isListening && handleKeyDown(e, action)}
              onClick={() => handleStartListen(action)}
              style={{
                padding: "4px 8px",
                background: isListening ? "var(--blue)" : "var(--bg-input)",
                border: `1px solid ${isListening ? "var(--blue)" : "var(--border)"}`,
                color: "var(--text)",
                fontFamily: "monospace",
                fontSize: 12,
                cursor: "pointer",
                minWidth: 60,
                textAlign: "center",
              }}
            />
            {isDuplicate && (
              <span style={{ fontSize: 10, color: "var(--red)" }}>Duplicate!</span>
            )}
          </div>
        );
      })}
    </div>
  );
}
```

```typescript
function useHotkeys() {
  const getKeybind = useKeybindStore((s) => s.getKeybind);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;
      const s = useUiStore.getState();

      const pauseKey = getKeybind("pause");
      const espionageKey = getKeybind("openEspionage");
      const diplomacyKey = getKeybind("openDiplomacy");
      const tradeKey = getKeybind("openTrade");
      const blueprintsKey = getKeybind("openBlueprints");
      const blackMarketKey = getKeybind("openBlackMarket");

      if (e.key === pauseKey) {
        s.setPaused(!s.paused);
        e.preventDefault();
      } else if (e.key.toLowerCase() === espionageKey) {
        s.toggleEspionagePanel();
      } else if (e.key.toLowerCase() === diplomacyKey) {
        s.toggleDiplomacyPanel();
      } else if (e.key.toLowerCase() === tradeKey) {
        s.toggleTradePanel();
      } else if (e.key.toLowerCase() === blueprintsKey) {
        s.toggleBlueprintShop();
      } else if (e.key.toLowerCase() === blackMarketKey) {
        s.toggleBlackMarket();
      }
    };

    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [getKeybind]);
```
