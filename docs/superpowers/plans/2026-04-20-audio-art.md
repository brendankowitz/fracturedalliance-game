  },
  {
    filename: "blueprint_purchased.wav",
    prompt: "Technology acquisition chime, research complete, triumphant, 1 second",
  },
  {
    filename: "independence_triggered.wav",
    prompt: "Dramatic orchestral sting, independence declaration, epic heroic fanfare, 2-3 seconds",
  },

async function generateAudio(filename, prompt) {
  console.log(`Generating ${filename}...`);

  const requestBody = {
    model: "speech-01",
    text: prompt,
    audio_format: "wav",
  };

  try {
    const response = await fetch(API_URL, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${API_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(requestBody),
    });

    if (!response.ok) {
      const error = await response.text();
      console.error(`Failed for ${filename}:`, response.status, error);
      return false;
    }

    const arrayBuffer = await response.arrayBuffer();
    const outputPath = path.join(projectRoot, "apps/web/public/audio/sfx", filename);

    await fs.mkdir(path.dirname(outputPath), { recursive: true });
    await fs.writeFile(outputPath, Buffer.from(arrayBuffer));

    console.log(`✓ Saved ${filename}`);
    return true;
  } catch (err) {
    console.error(`Error generating ${filename}:`, err.message);
    return false;
  }
}

async function main() {
  console.log("Starting SFX generation via Minimax API...\n");

  let successCount = 0;
  for (const spec of SFX_SPECS) {
    const success = await generateAudio(spec.filename, spec.prompt);
    if (success) successCount++;
    // Rate limit: small delay between requests
    await new Promise((r) => setTimeout(r, 500));
  }

  console.log(`\nGeneration complete: ${successCount}/${SFX_SPECS.length} files created`);
  process.exit(successCount === SFX_SPECS.length ? 0 : 1);
}

main();

## Section 4: Implement MusicPlayer Class

**File:** `apps/web/src/audio/MusicPlayer.ts`

**Instructions:**
- [ ] Create file `apps/web/src/audio/MusicPlayer.ts` with code below
- [ ] **Not a React component** — standalone TypeScript class using Web Audio API
- [ ] Manages 3 ambient tracks: exploration, conflict, endgame
- [ ] Implements crossfading between tracks
- [ ] Reads game state (no active combat, # of ticks to deadline)
- [ ] Autoplay on game start, pause on game pause

**Script Code:**

```typescript
export interface GameStateSnapshot {
  tick: number;
  deadlineTick: number;
  hasActiveCombat: boolean;
  isPaused: boolean;
}

type MusicTrack = "exploration" | "conflict" | "endgame";

export class MusicPlayer {
  private audioElements: Record<MusicTrack, HTMLAudioElement> = {
    exploration: new Audio("/audio/music/exploration.ogg"),
    conflict: new Audio("/audio/music/conflict.ogg"),
    endgame: new Audio("/audio/music/endgame.ogg"),
  };

  private currentTrack: MusicTrack | null = null;
  private isInitialized = false;
  private fadeOutInterval: number | null = null;
  private fadeInInterval: number | null = null;
  private readonly FADE_DURATION = 2000; // 2 second crossfade
  private readonly FADE_STEP = 50; // Update every 50ms

  constructor() {
    // Initialize all audio elements
    Object.values(this.audioElements).forEach((audio) => {
      audio.loop = true;
      audio.volume = 0;
      audio.preload = "auto";
    });
  }

  /**
   * Initialize autoplay after user interaction (browser autoplay policy)
   */
  init(): void {
    if (this.isInitialized) return;
    this.isInitialized = true;
    // Preload all tracks
    Object.values(this.audioElements).forEach((audio) => {
      audio.load();
    });
  }

  /**
   * Determine which track should play based on game state
   */
  private selectTrack(state: GameStateSnapshot): MusicTrack {
    const ticksRemaining = state.deadlineTick - state.tick;

    // Priority: endgame > conflict > exploration
    if (ticksRemaining < 200) {
      return "endgame";
    }
    if (state.hasActiveCombat) {
      return "conflict";
    }
    return "exploration";
  }

  /**
   * Update music based on current game state
   * Call this from your game loop (e.g., in renderLoop.ts)
   */
  update(state: GameStateSnapshot): void {
    if (!this.isInitialized) return;

    const targetTrack = this.selectTrack(state);

    // Pause all if game is paused
    if (state.isPaused) {
      Object.values(this.audioElements).forEach((audio) => {
        audio.pause();
      });
      return;
    }

    // No change needed
    if (targetTrack === this.currentTrack) {
      // Ensure playing if volume > 0
      if (this.audioElements[targetTrack].volume > 0) {
        this.audioElements[targetTrack].play().catch(() => {
          // Autoplay blocked or error, silently fail
        });
      }
      return;
    }

    // Track change: fade out current, fade in new
    this.fadeToTrack(targetTrack);
  }

  /**
   * Crossfade from current track to new track
   */
  private fadeToTrack(newTrack: MusicTrack): void {
    // Clear existing fade timers
    if (this.fadeOutInterval !== null) {
      clearInterval(this.fadeOutInterval);
      this.fadeOutInterval = null;
    }
    if (this.fadeInInterval !== null) {
      clearInterval(this.fadeInInterval);
      this.fadeInInterval = null;
    }

    const oldTrack = this.currentTrack;
    const newAudio = this.audioElements[newTrack];
    const steps = Math.ceil(this.FADE_DURATION / this.FADE_STEP);
    let step = 0;

    // Start new track from beginning
    newAudio.currentTime = 0;
    newAudio
      .play()
      .catch(() => {
        /* autoplay blocked */
      });

    // Fade in new track
    const fadeInId = setInterval(() => {
      step++;
      const progress = step / steps;
      newAudio.volume = progress * 0.7; // Target volume 0.7

      if (step >= steps) {
        clearInterval(fadeInId);
        this.fadeInInterval = null;
        newAudio.volume = 0.7;
      }
    }, this.FADE_STEP);

    this.fadeInInterval = fadeInId as any;

    // Fade out old track
    if (oldTrack) {
      step = 0;
      const oldAudio = this.audioElements[oldTrack];
      const fadeOutId = setInterval(() => {
        step++;
        const progress = 1 - step / steps;
        oldAudio.volume = Math.max(0, progress * 0.7);

        if (step >= steps) {
          clearInterval(fadeOutId);
          this.fadeOutInterval = null;
          oldAudio.pause();
          oldAudio.volume = 0;
        }
      }, this.FADE_STEP);

      this.fadeOutInterval = fadeOutId as any;
    }

    this.currentTrack = newTrack;
  }

  /**
   * Clean up resources
   */
  destroy(): void {
    if (this.fadeOutInterval !== null) clearInterval(this.fadeOutInterval);
    if (this.fadeInInterval !== null) clearInterval(this.fadeInInterval);
    Object.values(this.audioElements).forEach((audio) => {
      audio.pause();
      audio.src = "";
    });
    this.currentTrack = null;
    this.isInitialized = false;
  }
}
```

async function generatePortrait(raceId, raceName, baseDescription, mood, moodText) {
  const filename = `${raceId}-${mood}.png`;
  console.log(`Generating ${filename}...`);

  const prompt = `${baseDescription}, ${moodText}, realistic portrait, 512x512, centered face, sci-fi style`;

  const requestBody = {
    model: "image-01",
    prompt: prompt,
    image_format: "png",
    style: "realistic",
    width: 512,
    height: 512,
  };

  try {
    const response = await fetch(API_URL, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${API_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(requestBody),
    });

    if (!response.ok) {
      const error = await response.text();
      console.error(`Failed for ${filename}:`, response.status, error);
      return false;
    }

    const data = await response.json();
    if (!data.file_id) {
      console.error(`No file_id in response for ${filename}`);
      return false;
    }

    // Minimax returns file_id; fetch the actual image
    const imageUrl = `https://api.minimax.chat/v1/files/${data.file_id}`;
    const imageResponse = await fetch(imageUrl, {
      headers: {
        Authorization: `Bearer ${API_KEY}`,
      },
    });

    if (!imageResponse.ok) {
      console.error(`Failed to fetch image for ${filename}:`, imageResponse.status);
      return false;
    }

    const imageBuffer = await imageResponse.arrayBuffer();
    const outputPath = path.join(projectRoot, "apps/web/public/assets/ambassadors", filename);

    await fs.mkdir(path.dirname(outputPath), { recursive: true });
    await fs.writeFile(outputPath, Buffer.from(imageBuffer));

    console.log(`✓ Saved ${filename}`);
    return true;
  } catch (err) {
    console.error(`Error generating ${filename}:`, err.message);
    return false;
  }
}

async function main() {
  console.log("Starting ambassador portrait generation via Minimax API...\n");

  let successCount = 0;
  const total = RACES.length * MOODS.length;

  for (const race of RACES) {
    for (const mood of MOODS) {
      const success = await generatePortrait(
        race.id,
        race.name,
        race.description,
        mood.mood,
        mood.text
      );
      if (success) successCount++;
      // Rate limit: small delay between requests
      await new Promise((r) => setTimeout(r, 1000));
    }
  }

  console.log(`\nGeneration complete: ${successCount}/${total} portraits created`);
  process.exit(successCount === total ? 0 : 1);
}

## Notes

- **API Rate Limits:** Minimax may have rate limits. Adjust `FADE_STEP` in generation scripts if needed (currently 500ms delay between API calls for SFX, 1000ms for portraits).
- **Audio Autoplay:** Browsers restrict autoplay. MusicPlayer requires user interaction first (click, keypress) before audio plays. Call `musicPlayer.init()` on first interaction.
- **Browser Compatibility:** Web Audio API + HTMLAudioElement supported in all modern browsers (Chrome, Firefox, Safari, Edge).
- **Volume Levels:** SFX at 0.5 (default), Independence trigger at 0.8 (dramatic), MusicPlayer at 0.7 (background). Adjust to taste.
- **Cooldown:** Existing `playSound()` in `audio.ts` has 3s cooldown to prevent spam. This is intentional for UI sounds.
